# RAGBase — Deployment Guide
# ─────────────────────────────────────────────────────────────────────────────

# 1. Provision Ubuntu 22.04 droplet on DigitalOcean (4GB RAM / 80GB SSD recommended)

# 2. Point your domain DNS A record → droplet IP
#    (Do this before starting — Caddy needs DNS to propagate for TLS provisioning)

# 3. Clone and run setup (takes ~15 minutes, mostly Whisper model download)
git clone https://github.com/yourname/ragbase /tmp/ragbase
cd /tmp/ragbase
sudo bash scripts/setup.sh

# 4. Fill in credentials in /opt/ragbase/.env
#    MCP_BEARER_TOKEN and ENCRYPTION_KEY are already generated by setup.sh.
#    You need to add:
#      DOMAIN                — your domain, e.g. ragbase.yourdomain.com
#      DROPBOX_APP_KEY       — from dropbox.com/developers
#      DROPBOX_APP_SECRET    — from dropbox.com/developers
#      DROPBOX_REFRESH_TOKEN — run scripts/dropbox_auth.py on your laptop first
#      ANTHROPIC_API_KEY     — optional, for auto-tagging
sudo nano /opt/ragbase/.env

# 4a. Generate Dropbox credentials (run this on your laptop, not the server):
#     pip install dropbox
#     python scripts/dropbox_auth.py
#     Then paste the three DROPBOX_* values into /opt/ragbase/.env

# 5. Install and link the Dropbox daemon on the droplet (syncs your files):
cd ~ && wget -O dropbox.py 'https://www.dropbox.com/download?dl=packages/dropbox.py'
python3 dropbox.py start
#    Follow the prompts to link your account.

# 6. Start everything
sudo systemctl start ragbase-api ragbase-mcp caddy

# 7. Verify
curl https://YOUR_DOMAIN/health
#    Expected: {"status":"ok","total_documents":0,"total_chunks":0}

# 8. Drop a file into your Dropbox sync folder.
#    The scanner runs every 5 minutes — check the dashboard for progress:
#    https://YOUR_DOMAIN/

# ─────────────────────────────────────────────────────────────────────────────
# Useful commands
# ─────────────────────────────────────────────────────────────────────────────

# Service status
systemctl status ragbase-api ragbase-mcp caddy

# Live logs
journalctl -u ragbase-api -f
journalctl -u ragbase-mcp -f

# Trigger a manual scan immediately (no need to wait 5 minutes)
curl -X POST https://YOUR_DOMAIN/api/scan \
     -H "Authorization: Bearer YOUR_MCP_BEARER_TOKEN"

# Trigger a manual backup
curl -X POST https://YOUR_DOMAIN/api/backup \
     -H "Authorization: Bearer YOUR_MCP_BEARER_TOKEN"

# Restore a backup
python scripts/restore_backup.py \
    --backup /opt/ragbase/backups/ragbase_2024-01-15_02-00.db.enc
